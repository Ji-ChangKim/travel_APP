// 1. Button Radius & Padding Specification
.btn-primary-lg {
  height: 52px; border-radius: 9999px; /* rounded-full */
  background-color: var(--color-primary); /* #ff6b57 */
  padding-left: 2rem; padding-right: 2rem;
  font-family: 'Plus Jakarta Sans'; font-size: 15px; font-weight: 600;
}

// 2. Input Field Core Token
.input-standard {
  height: 44px; border-radius: 8px; /* ROUND_EIGHT */
  background-color: var(--color-surface-container-low); /* #f8f3ed */
  color: var(--color-on-surface); /* #1d1b18 */
}

// 3. Card Container Geometry
.surface-card {
  border-radius: 16px; /* rounded-xl */
  background-color: var(--color-surface-container-lowest); /* #ffffff */
  box-shadow: 0 4px 16px -2px rgba(32, 50, 71, 0.05);
}
